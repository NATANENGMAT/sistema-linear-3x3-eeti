Script SQL para o Supabase

Copie o comando abaixo e cole no SQL Editor do seu painel do Supabase (clique aqui para abrir). Isso cria a tabela necessária para receber as entregas dos alunos: 

create table if not exists public.entregas (
  id bigint generated by default as identity primary key,
  student_name text not null,
  score integer not null,
  total integer not null default 8,
  answers jsonb,
  submitted_at text,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Habilitar acesso público para envio e visualização
alter table public.entregas enable row level security;

create policy "Permitir leitura publica" 
  on public.entregas for select using (true);

create policy "Permitir envio publico" 
  on public.entregas for insert with check (true);
